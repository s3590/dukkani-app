dukkani-app/

│   ├── middlewares.py          # أنظمة الحماية، المصادقة (JWT )، وصائد الأخطاء
│   ├── views.py                # عرض صفحات HTML (مع نظام التخزين في الذاكرة Cache)
│   └── routes/                 # مسارات الـ API مقسمة حسب الوظيفة
│       ├── admin.py            # مسارات لوحة القيادة العليا (God Mode)
│       ├── customer.py         # مسارات تطبيق الزبون والعائلة
│       ├── manage.py           # مسارات إدارة البقالة (المنتجات، العمال، التقارير)
│       ├── misc.py             # مسارات متنوعة (الصور، الإشعارات، الأخطاء)
│       ├── pos.py              # مسارات الكاشير السريع
│       └── websockets.py       # الرادار الحي (WebSockets & Redis)
│
├── bot/                        # 🤖 بوت التليجرام (Telegram Bot)
│   ├── setup.py                # إعداد البوت وصائد الأخطاء الشامل
│   ├── keyboards.py            # لوحات المفاتيح (الأزرار السفلية والشفافة)
│   ├── states.py               # حالات البوت (FSM) للتنقل بين الخطوات
│   └── handlers/               # أوامر البوت مقسمة حسب المستخدم
│       ├── admin.py            # أوامر المدير العام (فورمات، إذاعة، نسخ احتياطي)
│       ├── store.py            # أوامر صاحب البقالة (قبول طلبات، إضافة منتجات)
│       └── worker.py           # أوامر العمال (تسجيل الدخول بالكود السري)
│
├── services/                   # ⚙️ الخدمات المركزية (Core Services)
│   ├── accounting.py           # المطبخ المركزي المالي (معالجة الطلبات والديون بأمان)
│   └── notifications.py        # محرك إشعارات Firebase (Push Notifications)
│
├── static/                     # 🖼️ الملفات الثابتة
│   ├── logo_customer.png       # شعار التطبيق الافتراضي
│   └── manifest.json           # ملف تعريف تطبيق الويب (PWA)
│
├── templates/                  # 🖥️ واجهات المستخدم (Frontend HTML/JS)
│   ├── index.html              # تطبيق الزبون والمندوب
│   ├── pos.html                # شاشة الكاشير السريع (Bento Box)
│   ├── manage.html             # لوحة إدارة البقالة
│   └── admin.html              # غرفة عمليات القيادة العليا
│
├── .env                        # 🔐 المتغيرات البيئية (لا يُرفع للعلن - يحتوي على الأسرار)
├── config.py                   # 🛠️ ملف الإعدادات (يقرأ من .env ويهيئ المتغيرات)
├── database.py                 # 🗄️ هيكل قاعدة البيانات (PostgreSQL) والمنظف التلقائي
├── firebase-messaging-sw.js    # 🔔 عامل الخدمة (Service Worker) لاستقبال الإشعارات في الخلفية
├── requirements.txt            # 📦 قائمة المكتبات المطلوبة لتشغيل المشروع
├── run.py                      # 🚀 محرك الإقلاع الرئيسي للسيرفر
│
# 👇 ملفات النشر السحابي (Deployment) 👇
├── Dockerfile                  # 🐳 ملف بناء حاوية التطبيق
└── docker-compose.yml          # 🐙 ملف تشغيل السيرفر، قاعدة البيانات، ورديس معاً
